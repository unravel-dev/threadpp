cmake_minimum_required(VERSION 3.14)

set(THREADPP_MAIN_PROJECT ON)
if(DEFINED PROJECT_SOURCE_DIR)
    set(THREADPP_MAIN_PROJECT OFF)
endif()

project(threadpp)
include(CTest)

set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake/utility")

option(BUILD_THREADPP_SHARED "Build as a shared library." ${THREADPP_MAIN_PROJECT})
option(BUILD_THREADPP_TESTS "Build the tests" ${THREADPP_MAIN_PROJECT})
option(BUILD_THREADPP_WITH_CODE_STYLE_CHECKS "Build with code style checks." ${THREADPP_MAIN_PROJECT})

if(BUILD_THREADPP_TESTS)
	if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
		set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)
	endif()
	if(NOT CMAKE_ARCHIVE_OUTPUT_DIRECTORY)
		set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/lib)
	endif()
	if(NOT CMAKE_LIBRARY_OUTPUT_DIRECTORY)
		set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/lib)
	endif()
endif()

if(BUILD_THREADPP_SHARED)
	set(BUILD_SHARED_LIBS ON)
endif()

if(BUILD_THREADPP_WITH_CODE_STYLE_CHECKS)
	set(BUILD_WITH_CODE_STYLE_CHECKS ON)
endif()

# Sanitizers for the library and its tests: "thread" (ThreadSanitizer, Linux and macOS) or
# "address;undefined". Built on its own, BUILD_WITH_SANITIZERS (the *-san presets) means the latter.
set(THREADPP_SANITIZE "" CACHE STRING "Sanitizers to build with: thread, or address;undefined. Empty for none.")
if(THREADPP_MAIN_PROJECT AND BUILD_WITH_SANITIZERS AND NOT THREADPP_SANITIZE)
	set(THREADPP_SANITIZE "address;undefined")
endif()
if(THREADPP_SANITIZE)
	if("thread" IN_LIST THREADPP_SANITIZE AND "address" IN_LIST THREADPP_SANITIZE)
		message(FATAL_ERROR "THREADPP_SANITIZE: thread and address cannot be combined.")
	endif()
	if(MSVC)
		if(NOT THREADPP_SANITIZE STREQUAL "address")
			message(FATAL_ERROR "THREADPP_SANITIZE: MSVC has only the address sanitizer.")
		endif()
		add_compile_options(/fsanitize=address /Zi)
	else()
		string(REPLACE ";" "," threadpp_sanitizers "${THREADPP_SANITIZE}")
		add_compile_options(-fsanitize=${threadpp_sanitizers} -fno-omit-frame-pointer -g)
		add_link_options(-fsanitize=${threadpp_sanitizers})
	endif()
	message(STATUS "threadpp: building with sanitizers ${THREADPP_SANITIZE}")
endif()

add_subdirectory(threadpp)

if(BUILD_THREADPP_TESTS)
	add_subdirectory(3rdparty)
	add_subdirectory(tests)
endif()
